## 3. Código Fonte Principal (Implementação Concreta)
O código principal `app.py` implementará um CRUD mínimo em SQLite e um endpoint de status de auditoria de exames.

```python
# app.py
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import sqlite3
from datetime import datetime

app = FastAPI()

# Configuração do banco SQLite
DATABASE = "exames.db"

# Criação da tabela 'exames' se não existir
def init_db():
    conn = sqlite3.connect(DATABASE)
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS exames (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            paciente TEXT NOT NULL,
            exame TEXT NOT NULL,
            data TEXT NOT NULL,
            status TEXT NOT NULL
        )
    ''')
    conn.commit()
    conn.close()

# Inicializa o banco de dados
init_db()

# Modelo Pydantic para os dados do exame
class Exame(BaseModel):
    paciente: str
    exame: str
    data: str
    status: str

# Endpoint para criar um novo exame
@app.post("/exames/", response_model=Exame)
def create_exame(exame: Exame):
    conn = sqlite3.connect(DATABASE)
    cursor = conn.cursor()
    cursor.execute('''
        INSERT INTO exames (paciente, exame, data, status)
        VALUES (?, ?, ?, ?)
    ''', (exame.paciente, exame.exame, exame.data, exame.status))
    conn.commit()
    conn.close()
    return exame

# Endpoint para obter todos os exames
@app.get("/exames/", response_model=list[Exame])
def get_exames():
    conn = sqlite3.connect(DATABASE)
    cursor = conn.cursor()
    cursor.execute('SELECT * FROM exames')
    rows = cursor.fetchall()
    conn.close()
    return [Exame(id=row[0], paciente=row[1], exame=row[2], data=row[3], status=row[4]) for row in rows]

# Endpoint para obter um exame por ID
@app.get("/exames/{exame_id}", response_model=Exame)
def get_exame(exame_id: int):
    conn = sqlite3.connect(DATABASE)
    cursor = conn.cursor()
    cursor.execute('SELECT * FROM exames WHERE id = ?', (exame_id,))
    row = cursor.fetchone()
    conn.close()
    if row:
        return Exame(id=row[0], paciente=row[1], exame=row[2], data=row[3], status=row[4])
    else:
        raise HTTPException(status_code=404, detail="Exame não encontrado")

# Endpoint para atualizar um exame
@app.put("/exames/{exame_id}", response_model=Exame)
def update_exame(exame_id: int, exame: Exame):
    conn = sqlite3.connect(DATABASE)
    cursor = conn.cursor()
    cursor.execute('''
        UPDATE exames
        SET paciente = ?, exame = ?, data = ?, status = ?
        WHERE id = ?
    ''', (exame.paciente, exame.exame, exame.data, exame.status, exame_id))
    conn.commit()
    conn.close()
    return exame

# Endpoint para deletar um exame
@app.delete("/exames/{exame_id}", response_model=Exame)
def delete_exame(exame_id: int):
    conn = sqlite3.connect(DATABASE)
    cursor = conn.cursor()
    cursor.execute('DELETE FROM exames WHERE id = ?', (exame_id,))
    conn.commit()
    conn.close()
    return {"id": exame_id}

# Endpoint para auditoria de exames
@app.get("/audit/", response_model=dict)
def audit_exames():
    conn = sqlite3.connect(DATABASE)
    cursor = conn.cursor()
    cursor.execute('SELECT COUNT(*) FROM exames WHERE status = "pendente"')
    pending_count = cursor.fetchone()[0]
    cursor.execute('SELECT COUNT(*) FROM exames WHERE status = "concluido"')
    completed_count = cursor.fetchone()[0]
    conn.close()
    return {
        "total_exames": pending_count + completed_count,
        "exames_pendentes": pending_count,
        "exames_concluidos": completed_count
    }
```
